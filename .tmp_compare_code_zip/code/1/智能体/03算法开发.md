## 完整代码实现
```python
import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
from typing import List

# 假设 SampleData 数据结构（与分布式框架对齐）
class SampleData:
    """训练样本数据结构，由环境/采样模块生成"""
    def __init__(self, state, action, reward, next_state, done, log_prob=None, value=None):
        self.state = state          # 状态
        self.action = action        # 动作
        self.reward = reward        # 奖励
        self.next_state = next_state# 下一状态
        self.done = done            # 终止标志
        self.log_prob = log_prob    # 动作对数概率（PPO/A2C使用）
        self.value = value          # 价值估计（PPO/A2C使用）

class BaseRLAlgorithm:
    """强化学习算法基类，实现核心learn接口"""
    def __init__(self, model, lr=1e-3, gamma=0.99, device="cpu"):
        """
        初始化算法
        :param model: 神经网络模型
        :param lr: 学习率
        :param gamma: 折扣因子
        :param device: 运行设备 cpu/cuda
        """
        self.model = model.to(device)
        self.optimizer = optim.Adam(self.model.parameters(), lr=lr)
        self.gamma = gamma
        self.device = device
        self.loss_fn = nn.MSELoss()  # 基础损失函数，可根据算法替换

    def _parse_sample_data(self, list_sample_data: List[SampleData]):
        """
        解析样本数据，转换为模型可训练的张量
        :return: 解析后的批量数据 (states, actions, rewards, next_states, dones)
        """
        states = []
        actions = []
        rewards = []
        next_states = []
        dones = []
        log_probs = []
        values = []

        for sample in list_sample_data:
            states.append(sample.state)
            actions.append(sample.action)
            rewards.append(sample.reward)
            next_states.append(sample.next_state)
            dones.append(sample.done)
            if sample.log_prob is not None:
                log_probs.append(sample.log_prob)
            if sample.value is not None:
                values.append(sample.value)

        # 转换为张量并迁移到指定设备
        states = torch.tensor(np.array(states), dtype=torch.float32).to(self.device)
        actions = torch.tensor(np.array(actions), dtype=torch.long).to(self.device)
        rewards = torch.tensor(np.array(rewards), dtype=torch.float32).to(self.device)
        next_states = torch.tensor(np.array(next_states), dtype=torch.float32).to(self.device)
        dones = torch.tensor(np.array(dones), dtype=torch.float32).to(self.device)
        
        log_probs = torch.tensor(np.array(log_probs), dtype=torch.float32).to(self.device) if log_probs else None
        values = torch.tensor(np.array(values), dtype=torch.float32).to(self.device) if values else None

        return states, actions, rewards, next_states, dones, log_probs, values

    def learn(self, list_sample_data: List[SampleData]):
        """
        实现强化学习算法核心方法
        :param list_sample_data: 训练样本(SampleData)列表
        :return: 训练损失 loss
        """
        # 1. 梯度清零（必须步骤，防止梯度累加）
        self.optimizer.zero_grad()

        # 2. 解析样本数据
        states, actions, rewards, next_states, dones, log_probs, values = self._parse_sample_data(list_sample_data)

        # ===================== 核心算法实现 =====================
        # 示例：DQN 算法实现（价值-based 强化学习）
        # 1. 计算当前Q值
        current_q = self.model(states).gather(1, actions.unsqueeze(1)).squeeze(1)
        
        # 2. 计算目标Q值
        with torch.no_grad():
            next_q = self.model(next_states).max(1)[0]
            target_q = rewards + self.gamma * next_q * (1 - dones)
        
        # 3. 计算损失
        loss = self.loss_fn(current_q, target_q)
        # ======================================================

        # 3. 反向传播计算梯度
        loss.backward()

        # 4. 优化器更新模型参数
        self.optimizer.step()

        return loss.item()

# 示例：PPO 算法扩展（策略梯度-based 强化学习，可直接替换learn函数）
class PPOAlgorithm(BaseRLAlgorithm):
    def learn(self, list_sample_data: List[SampleData], clip_epsilon=0.2):
        """PPO 算法实现"""
        self.optimizer.zero_grad()
        states, actions, rewards, next_states, dones, old_log_probs, old_values = self._parse_sample_data(list_sample_data)

        # 计算优势函数
        with torch.no_grad():
            next_value = self.model.get_value(next_states)
            target_value = rewards + self.gamma * next_value * (1 - dones)
            advantage = target_value - old_values

        # 计算新的动作概率
        new_log_probs = self.model.get_log_prob(states, actions)
        
        # PPO 裁剪损失
        ratio = torch.exp(new_log_probs - old_log_probs)
        surr1 = ratio * advantage
        surr2 = torch.clamp(ratio, 1 - clip_epsilon, 1 + clip_epsilon) * advantage
        loss = -torch.min(surr1, surr2).mean()

        # 梯度更新
        loss.backward()
        self.optimizer.step()

        return loss.item()
```

## 核心函数说明
### `learn(self, list_sample_data)`
**功能**：强化学习算法核心训练函数，完成**数据解析→损失计算→梯度反向传播→参数更新**全流程。

| 参数名 | 类型 | 说明 |
|--------|------|------|
| list_sample_data | `List[SampleData]` | 训练样本列表，每个元素包含状态、动作、奖励、终止标志等完整训练信息 |

### 关键执行步骤
1. **梯度清零**：`optimizer.zero_grad()` 清除上一轮训练的梯度，避免梯度累加
2. **数据解析**：将样本列表转换为 PyTorch 张量，适配模型训练
3. **算法核心逻辑**：根据所选强化学习算法（DQN/PPO/A2C等）计算损失
4. **反向传播**：`loss.backward()` 自动计算模型参数梯度
5. **参数更新**：`optimizer.step()` 通过优化器更新网络权重
